# dns-blocklist: telemetry, analytics, ad attribution and crash reporting missed by the upstream lists.
# See README.md. A registrable domain blocks all of its subdomains; a deeper name blocks only itself.

# --- App SDKs (mobile) ---
# AppsFlyer attribution SDK: apps use random per-app subdomains (xxxx.inapps / launches / dlsdk / skadsdk)
appsflyersdk.com
# Firebase Crashlytics crash reporting and settings
crashlytics.com
# Firebase / Google Analytics for apps (also used by smart TV apps)
app-measurement.com
# iOS app analytics
app-analytics-services.com
# Adjust analytics (app.adjust.com redirects are intentionally not blocked)
analytics.adjust.com
# Firebase / Google SDK client logging (firebaseremoteconfig and firebaseappcheck are app features, not blocked)
firebaselogging-pa.googleapis.com
# Taobao push SDK delivery reports
agoo-report.m.taobao.com
# Jin10 app usage statistics
big-data-statistics.jin10.com
# Jin10 ad API (ad slots, ad click stats; quotes and news use other hosts)
jad-api.jin10.com

# --- Web analytics ---
# Hotjar session recording and heatmaps
hotjar.com
hotjar.io
# New Relic browser agent
js-agent.newrelic.com
# Twitter / X analytics
analytics.twitter.com
# Judge.me review widget tracking
tracking.aws.judge.me
# Ad conversion pixel
pixel-for-openai-ads.fly.dev
# Datadog browser RUM intake (US5 site)
browser-intake-us5-datadoghq.com
# Cloudflare Web Analytics beacon (whole domain, including static.cloudflareinsights.com)
cloudflareinsights.com
# GitHub web analytics collector
collector.github.com

# --- Vendor telemetry ---
# Farfetch app telemetry
telemetry.farfetch.net
# Uniswap web app metrics
metrics.interface.gateway.uniswap.org
# Amazon Shopping app behavioral telemetry (whole domain, including api.mshop.bdtelemetry.amazon)
bdtelemetry.amazon
# WeChat client reports (only this name; sgshort / sglong / pay hosts are the app itself)
sgreport.wechat.com
# Graylog server telemetry (can also be turned off with telemetry_enabled = false)
telemetry.graylog.cloud
# Thunderbird telemetry (Glean pings)
incoming-telemetry.thunderbird.net
# Reolink app telemetry
telemetric-apis.reolink.com
